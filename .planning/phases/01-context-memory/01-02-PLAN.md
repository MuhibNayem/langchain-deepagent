```python
def list(self) -> list[dict]:
    """Return all sessions sorted by last_accessed descending."""
    index = self._load_index()
    sessions = list(index["sessions"].values())
    return sorted(sessions, key=lambda s: s["last_accessed"], reverse=True)

def cleanup(self, max_age_days: int = 30) -> list[str]:
    """Remove sessions older than max_age_days. Returns removed thread_ids."""
    index = self._load_index()
    cutoff = datetime.utcnow() - timedelta(days=max_age_days)
    removed = []
    for thread_id, entry in index["sessions"].items():
        last_accessed = datetime.fromisoformat(entry["last_accessed"])
        if last_accessed < cutoff:
            # Remove JSONL file
            session_file = self._session_dir / f"{thread_id}.jsonl"
            if session_file.exists():
                session_file.unlink()
            del index["sessions"][thread_id]
            removed.append(thread_id)
    self._save_index(index)
    return removed
```
