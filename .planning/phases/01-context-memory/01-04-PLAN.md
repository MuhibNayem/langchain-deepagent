```python
from dataclasses import dataclass

@dataclass
class CompressionResult:
    compressed_messages: list[dict]
    original_count: int
    compressed_count: int
    compression_ratio: float  # compressed_count / original_count
    token_budget_used: int

class ContextCompactor:
    def __init__(
        self,
        max_tokens: int = 128000,
        recent_ratio: float = 0.7,  # 70% recent, 30% compressed
        compression_factor: float = 0.5  # compress older by 50%
    ):
        self.max_tokens = max_tokens
        self.recent_ratio = recent_ratio
        self.compression_factor = compression_factor

    def compact(self, messages: list[dict], prefix: str = "") -> CompressionResult:
        """Compress message list to fit within max_tokens."""
        # Calculate available budget for messages (exclude prefix)
        available = self.max_tokens - self._estimate_tokens(prefix)

        # Recent-biased: keep recent turns rich, compress older
        total = len(messages)
        recent_count = int(total * self.recent_ratio)
        compressed_count = total - recent_count

        result_messages = []
        tokens_used = 0

        # Add recent messages (rich)
        for msg in messages[-recent_count:]:
            msg_tokens = self._estimate_tokens(msg)
            if tokens_used + msg_tokens > available:
                break  # budget exhausted
            result_messages.append(msg)
            tokens_used += msg_tokens

        # Add compressed older messages
        for msg in messages[:-recent_count]:
            compressed = self._compress_message(msg)
            comp_tokens = self._estimate_tokens(compressed)
            if tokens_used + comp_tokens > available:
                # Apply quality fallback: summarize instead of drop
                compressed = self._summarize_messages_above(msg)
                comp_tokens = self._estimate_tokens(compressed)
                if tokens_used + comp_tokens > available:
                    continue  # drop if still over
            result_messages.append(compressed)
            tokens_used += comp_tokens

        return CompressionResult(...)

    def _compress_message(self, msg: dict) -> dict:
        """Apply compression to single message."""
        # Reduce content, keep structure
        return {
            "role": msg.get("role"),
            "content": self._truncate_content(msg.get("content", "")),
            "timestamp": msg.get("timestamp")
        }

    def _truncate_content(self, content: str, max_chars: int = 500) -> str:
        """Truncate content to max_chars."""
        if len(content) <= max_chars:
            return content
        return content[:max_chars] + "... [compressed]"

    def _summarize_messages_above(self, msg: dict) -> dict:
        """Fallback: summarize rather than drop."""
        return {
            "role": msg.get("role"),
            "content": f"[Earlier conversation summarized] {msg.get('content', '')[:200]}...",
            "timestamp": msg.get("timestamp")
        }

    def _estimate_tokens(self, obj: dict | str) -> int:
        """Rough token estimation (1 token ≈ 4 chars)."""
        text = obj if isinstance(obj, str) else str(obj)
        return len(text) // 4
```

```python
class ContextCompactor:
    def __init__(self, ..., dedup_window: int = 10):
        self.dedup_window = dedup_window  # turns within which to dedup
        self._seen_files: dict[str, str] = {}  # path → content_hash
        self._turn_index: int = 0

    def deduplicate(self, messages: list[dict]) -> list[dict]:
        """Remove file reads with duplicate content within window."""
        self._turn_index += 1
        result = []
        for msg in messages:
            if self._is_file_read(msg):
                content = self._extract_file_content(msg)
                content_hash = hashlib.sha256(content.encode()).hexdigest()
                path = self._extract_file_path(msg)

                if path in self._seen_files:
                    if self._seen_files[path] == content_hash:
                        # Duplicate within window — replace with reference
                        result.append({
                            "role": msg.get("role"),
                            "content": f"[File {path} unchanged — see above]",
                            "is_dedup_ref": True
                        })
                        continue
                    else:
                        # Content changed — update hash
                        self._seen_files[path] = content_hash
                else:
                    self._seen_files[path] = content_hash

                # Prune old entries outside window
                if len(self._seen_files) > self.dedup_window:
                    # Remove oldest entries
                    self._seen_files = dict(list(self._seen_files.items())[-self.dedup_window:])

            result.append(msg)
        return result

    def _is_file_read(self, msg: dict) -> bool:
        """Detect if message is a file read tool result."""
        return msg.get("type") == "tool_result" and "read_file" in msg.get("name", "")

    def _extract_file_content(self, msg: dict) -> str:
        """Extract content from file read message."""
        return msg.get("content", "")

    def _extract_file_path(self, msg: dict) -> str:
        """Extract file path from message."""
        # Parse from tool call arguments
        return msg.get("path", msg.get("file_path", ""))
```

```python
class SessionStore:
    def __init__(self, ..., compactor: ContextCompactor | None = None):
        self.compactor = compactor or ContextCompactor()

    def auto_compact(self, thread_id: str) -> None:
        """Run compaction on each agent turn (per D-04)."""
        session = self.get_session(thread_id)
        messages = session.full_transcript.get_all_messages()
        prefix = self._prompt_builder.get_prefix() if self._prompt_builder else ""
        result = self.compactor.compact(messages, prefix)

        # If compaction happened, update working memory with stats
        if result.compression_ratio < 1.0:
            session.working_memory.add_note(
                f"Compacted {result.original_count} → {result.compressed_count} messages "
                f"({result.compression_ratio:.0%} ratio, {result.token_budget_used} tokens)"
            )
```
