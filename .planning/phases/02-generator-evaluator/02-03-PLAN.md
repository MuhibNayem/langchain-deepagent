from luminamind.evaluator.criteria import CodeCriteria

class CodeEvaluator(CodeCriteria):
    """Code quality evaluator per GE-06.

    Scores: correctness, maintainability, performance, security.
    """

    def evaluate(self, artifact) -> dict:
        """Evaluate code artifact.

        Returns dict with:
            - correctness_score: float (0-100)
            - maintainability_score: float (0-100)
            - performance_score: float (0-100)
            - security_score: float (0-100)
            - overall_score: float
            - issues: list[dict]
            - strengths: list[str]
            - recommendations: list[dict]
        """
        correctness = self._score_correctness(artifact)
        maintainability = self._score_maintainability(artifact)
        performance = self._score_performance(artifact)
        security = self._score_security(artifact)

        overall = (
            correctness["score"] * 0.30 +
            maintainability["score"] * 0.25 +
            performance["score"] * 0.20 +
            security["score"] * 0.25
        )

        return {
            "correctness_score": correctness["score"],
            "maintainability_score": maintainability["score"],
            "performance_score": performance["score"],
            "security_score": security["score"],
            "overall_score": overall,
            "issues": correctness["issues"] + maintainability["issues"],
            "strengths": [],
            "recommendations": [],
        }

    def _score_correctness(self, artifact) -> dict:
        """Score correctness: bug presence, logic errors, edge cases."""
        issues = []
        # Static analysis for:
        # - Infinite loops (while True without break)
        # - Null dereferences (undefined variable access)
        # - Type errors (wrong type passed)
        # - Logic errors (off-by-one, incorrect conditionals)
        score = 100 - (len(issues) * 10)
        return {"score": max(0, score), "issues": issues}

    def _score_maintainability(self, artifact) -> dict:
        """Score maintainability: complexity, coupling, naming."""
        issues = []
        # Metrics:
        # - Cyclomatic complexity (>10 = warning, >20 = critical)
        # - Function length (>50 lines = warning, >100 = critical)
        # - Nesting depth (>4 = warning, >7 = critical)
        # - Naming conventions
        # - Magic numbers
        score = 100 - (len(issues) * 10)
        return {"score": max(0, score), "issues": issues}
```

import pytest
from luminamind.evaluator.code import CodeEvaluator

CORRECT_CODE = "def add(a, b): return a + b"
BUGGY_CODE = "def divide(a, b): return a / b  # Missing zero check"
COMPLEX_CODE = """
for i in range(n):
    for j in range(n):
        for k in range(n):
            print(i, j, k)
"""
INSECURE_CODE = "import os; os.system('rm -rf ' + user_input)"

class TestCodeEvaluator:
    def test_correctness_scoring(self):
        evaluator = CodeEvaluator()
        result = evaluator.evaluate(BUGGY_CODE)
        assert result["correctness_score"] < 100
        assert any("divide" in str(i) for i in result["issues"])

    def test_maintainability_scoring(self):
        evaluator = CodeEvaluator()
        result = evaluator.evaluate(COMPLEX_CODE)
        assert result["maintainability_score"] < 80

    def test_performance_scoring(self):
        evaluator = CodeEvaluator()
        result = evaluator.evaluate(COMPLEX_CODE)
        assert result["performance_score"] < 70

    def test_security_scoring(self):
        evaluator = CodeEvaluator()
        result = evaluator.evaluate(INSECURE_CODE)
        assert result["security_score"] < 50
        assert any("shell" in str(i).lower() for i in result["issues"])

    def test_overall_score_calculation(self):
        evaluator = CodeEvaluator()
        result = evaluator.evaluate(CORRECT_CODE)
        assert 0 <= result["overall_score"] <= 100
        # Weighted average check
        expected = (
            result["correctness_score"] * 0.30 +
            result["maintainability_score"] * 0.25 +
            result["performance_score"] * 0.20 +
            result["security_score"] * 0.25
        )
        assert abs(result["overall_score"] - expected) < 0.01
```
