    1. __init__(self, deep_agent: DeepAgent, output_dir: Path):
       - Initialize with DeepAgent to test
       - Set up fault injection hooks
       - output_dir for results
    2. run_scenario(self, scenario: ChaosScenario, task: str) -> ChaosResult:
       - Inject fault(s) as specified in scenario
       - Execute task through DeepAgent
       - Record what happened
       - Collect fault injection metrics
    3. run_suite(self, scenarios: list[ChaosScenario], task: str) -> ChaosSuiteResult:
       - Run multiple scenarios against same task
       - Compare behavior across scenarios
    4. _inject_fault(self, fault: FaultConfig) -> None:
       - Use unittest.mock to patch network/IO calls
       - Or modify environment variables
       - Or use fault-specific injection method
    5. _cleanup_faults(self) -> None:
       - Restore all patched functions
    
    ChaosResult dataclass:
    - scenario_id: str
    - task: str
    - fault_injected: bool
    - execution_time: float
    - outcome: "success" | "degraded" | "failed"
    - error_message: str | None
    - graceful_degradation: bool
    - metrics: dict
